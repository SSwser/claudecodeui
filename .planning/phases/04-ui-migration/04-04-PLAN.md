**Step 1 — Install Playwright (if not already installed):**
```bash
npm install --save-dev @playwright/test
npx playwright install chromium --with-deps
```

**Step 2 — Create `playwright.config.ts`:**
```typescript
import { defineConfig, devices } from '@playwright/test';

export default defineConfig({
  testDir: './tests',
  fullyParallel: false,
  forbidOnly: !!process.env.CI,
  retries: process.env.CI ? 1 : 0,
  reporter: 'html',
  use: {
    baseURL: process.env.BASE_URL || 'http://localhost:5173',
    trace: 'on-first-retry',
    screenshot: 'only-on-failure',
  },
  snapshotDir: './tests/visual/__snapshots__',
  snapshotPathTemplate: '{snapshotDir}/{testFilePath}/{arg}{ext}',
  projects: [
    {
      name: 'chromium',
      use: {
        ...devices['Desktop Chrome'],
        viewport: { width: 1280, height: 800 },
      },
    },
  ],
  // Webserver not auto-started — run `npm run dev` before visual tests
});
```

**Step 3 — Create `tests/visual/baseline.spec.ts`:**

```typescript
import { test, expect } from '@playwright/test';

// Phase 4 visual regression surface (per D-10, D-11, D-12)
// Covers: every page touched by Phase 4 migration + global shell surfaces

test.describe('Visual Regression — Phase 4 Migration', () => {
  test.beforeEach(async ({ page }) => {
    // Suppress any network errors from missing backend
    page.on('requestfailed', () => {});
  });

  test('Auth — login page (light)', async ({ page }) => {
    await page.goto('/login');
    await page.waitForLoadState('networkidle');
    await expect(page).toHaveScreenshot('auth-login-light.png', {
      fullPage: true,
      maxDiffPixelRatio: 0.02,
    });
  });

  test('Settings page (light)', async ({ page }) => {
    await page.goto('/settings');
    await page.waitForLoadState('networkidle');
    await expect(page).toHaveScreenshot('settings-light.png', {
      fullPage: true,
      maxDiffPixelRatio: 0.02,
    });
  });

  test('Sidebar shell — default state (light)', async ({ page }) => {
    await page.goto('/');
    await page.waitForLoadState('networkidle');
    await expect(page.locator('[data-testid="sidebar"]').first()).toHaveScreenshot(
      'sidebar-shell-light.png',
      { maxDiffPixelRatio: 0.02 },
    );
  });

  test('Chat view — shell layout (light)', async ({ page }) => {
    await page.goto('/');
    await page.waitForLoadState('networkidle');
    await expect(page).toHaveScreenshot('chat-shell-light.png', {
      fullPage: false,
      clip: { x: 0, y: 0, width: 1280, height: 800 },
      maxDiffPixelRatio: 0.02,
    });
  });

  test('Dialog component — open state (light)', async ({ page }) => {
    // Navigate to a page that has a dialog trigger, or inject a test page
    await page.goto('/?testDialog=1');
    await page.waitForLoadState('networkidle');
    // Attempt to open a dialog if trigger is present
    const trigger = page.locator('[data-testid="dialog-trigger"]').first();
    if (await trigger.count() > 0) {
      await trigger.click();
      await page.waitForTimeout(200);
    }
    await expect(page).toHaveScreenshot('dialog-open-light.png', {
      maxDiffPixelRatio: 0.02,
    });
  });

  test('Dark mode — login page', async ({ page }) => {
    await page.goto('/login');
    // Toggle dark mode via localStorage
    await page.evaluate(() => {
      document.documentElement.classList.add('dark');
      localStorage.setItem('theme', 'dark');
    });
    await page.waitForLoadState('networkidle');
    await expect(page).toHaveScreenshot('auth-login-dark.png', {
      fullPage: true,
      maxDiffPixelRatio: 0.02,
    });
  });
});
```

Note: Snapshots are intentionally gitignored on first run to avoid large binary files
in history. The CI workflow should generate and store them as artifacts.

**Note on `data-testid` attributes:** If pages do not yet have `data-testid="sidebar"`
or `data-testid="dialog-trigger"`, the corresponding test block will gracefully skip
that interaction (count() === 0). Add `data-testid` to sidebar and dialog trigger
elements as a non-breaking change (add only, do not restructure other components).

Add `data-testid="sidebar"` to the outermost container in the sidebar component
(find the file using pattern `src/components/sidebar/view/*.tsx`).
